👁 TripDetails VIEW

Joins Trip ⟵ Driver, Vehicle, Route to produce a consolidated view.

CREATE VIEW TripDetails AS
SELECT d.name AS driver, v.vehicle_no, r.source, r.destination, t.trip_date
FROM Trip t
JOIN Driver d ON t.driver_id = d.driver_id
JOIN Vehicle v ON t.vehicle_id = v.vehicle_id
JOIN Route r ON t.route_id = r.route_id;

📋 CREATE TABLES
-- Driver Table
CREATE TABLE Driver (
  driver_id INT PRIMARY KEY,
  name VARCHAR(50),
  license_no VARCHAR(20) UNIQUE,
  phone VARCHAR(15)
);

-- Vehicle Table
CREATE TABLE Vehicle (
  vehicle_id INT PRIMARY KEY,
  vehicle_no VARCHAR(20) UNIQUE,
  type VARCHAR(20),
  capacity INT CHECK (capacity > 0)
);

-- Route Table
CREATE TABLE Route (
  route_id INT PRIMARY KEY,
  source VARCHAR(50),
  destination VARCHAR(50),
  distance INT
);

-- Trip Table
CREATE TABLE Trip (
  trip_id INT PRIMARY KEY,
  driver_id INT,
  vehicle_id INT,
  route_id INT,
  trip_date DATE,
  FOREIGN KEY (driver_id) REFERENCES Driver(driver_id),
  FOREIGN KEY (vehicle_id) REFERENCES Vehicle(vehicle_id),
  FOREIGN KEY (route_id) REFERENCES Route(route_id)
);

-- Passenger Table
CREATE TABLE Passenger (
  passenger_id INT PRIMARY KEY,
  name VARCHAR(50),
  phone VARCHAR(15)
);

-- Booking Table
CREATE TABLE Booking (
  booking_id INT PRIMARY KEY,
  passenger_id INT,
  trip_id INT,
  seat_no INT,
  booking_date DATE,
  UNIQUE (trip_id, seat_no),
  FOREIGN KEY (passenger_id) REFERENCES Passenger(passenger_id),
  FOREIGN KEY (trip_id) REFERENCES Trip(trip_id)
);
